  <script>
    // ==========================================
    // DEVELOPER CONFIGURATION & BOUNDARIES
    // ==========================================
    const CONFIG = {
      NAME_X: 110,
      NAME_Y: 562.5,
      NAME_LINE_HEIGHT: 65, 
      BATCH_X: 110,
      BATCH_Y: 804,
      COLLEGE_X: 110,
      COLLEGE_Y: 846
    };

    // Developer boundary limits for photo dragging 
    const BOUNDS = {
      MIN_X: 150,  // How far left the user can drag
      MAX_X: 430,  // Threshold for right boundary limit
      MIN_Y: 100,  // How far up the user can drag
      MAX_Y: 245   // Threshold for bottom boundary limit
    };

    const canvas = document.getElementById('posterCanvas');
    const ctx = canvas.getContext('2d');

    // Inputs & Buttons
    const photoInput = document.getElementById('userPhoto');
    const recropBtn = document.getElementById('recropBtn');
    const zoomInput = document.getElementById('photoZoom');
    const resetBtn = document.getElementById('resetPhotoBtn');
    const nameInput = document.getElementById('userName');
    const batchInput = document.getElementById('userBatch');
    const collegeInput = document.getElementById('userCollege');
    const downloadBtn = document.getElementById('downloadBtn');

    // Cropper Modal Elements
    const cropperModal = document.getElementById('cropperModal');
    const cropTargetImage = document.getElementById('cropTargetImage');
    const applyCropBtn = document.getElementById('applyCropBtn');
    const cancelCropBtn = document.getElementById('cancelCropBtn');

    // State Variables
    let rawImageSrc = null;  // Original uncropped image data
    let userImage = null;    // Cropped HTML Image instance used for rendering
    let cropper = null;      // Cropper.js Instance

    let photoX = 580;       
    let photoY = 410;       
    let baseWidth = 440;    
    let baseHeight = 480;   
    let scale = 1.58;        

    // Drag-and-drop tracking states
    let isDragging = false;
    let dragStartX = 0;
    let dragStartY = 0;

    // Fixed Template Background Image
    const bgTemplateImage = new Image();
    bgTemplateImage.src = 'temp.svg'; 

    bgTemplateImage.onload = () => renderCanvas();

    // Helper function to smart-wrap name if length/words exceed limit (~11 chars)
    function getWrappedLines(text, maxLen = 11) {
      const words = text.trim().split(/\s+/);
      const lines = [];
      let currentLine = '';
      
      for (let word of words) {
        if ((currentLine + ' ' + word).trim().length <= maxLen || currentLine === '') {
          currentLine = currentLine ? currentLine + ' ' + word : word;
        } else {
          lines.push(currentLine);
          currentLine = word;
        }
      }
      if (currentLine) {
        lines.push(currentLine);
      }
      return lines;
    }

    function renderCanvas() {
      // Clear Canvas
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // 1. Draw Fixed Template Background
      ctx.drawImage(bgTemplateImage, 0, 0, canvas.width, canvas.height);

      // 2. Draw Cropped User Photo
      if (userImage) {
        const currentWidth = baseWidth * scale;
        const currentHeight = baseHeight * scale;

        // Enforce boundary constraints during render/drag
        if (photoX > BOUNDS.MAX_X) photoX = BOUNDS.MAX_X;
        if (photoY > BOUNDS.MAX_Y) photoY = BOUNDS.MAX_Y;
        if (photoX < BOUNDS.MIN_X) photoX = BOUNDS.MIN_X;
        if (photoY < BOUNDS.MIN_Y) photoY = BOUNDS.MIN_Y;

        ctx.drawImage(userImage, photoX, photoY, currentWidth, currentHeight);
      }

      // 3. Draw Dynamic Name 
      ctx.fillStyle = "#FFFFFF";
      ctx.font = "800 68px 'Inter', sans-serif";
      ctx.textAlign = "left";
      ctx.textBaseline = "top";
      
      const nameText = nameInput.value || "YOUR NAME";
      const nameLines = getWrappedLines(nameText, 11);
      
      nameLines.forEach((line, index) => {
        ctx.fillText(line, CONFIG.NAME_X, CONFIG.NAME_Y + (index * CONFIG.NAME_LINE_HEIGHT));
      });

      // 4. Draw Dynamic Batch 
      ctx.font = "400 32px 'Inter', sans-serif";
      const batchText = batchInput.value || "Batch Year";
      ctx.fillText(batchText, CONFIG.BATCH_X, CONFIG.BATCH_Y);

      // 5. Draw Dynamic College Name 
      ctx.font = "500 30px 'Inter', sans-serif";
      const collegeText = collegeInput.value || "College Name";
      ctx.fillText(collegeText, CONFIG.COLLEGE_X, CONFIG.COLLEGE_Y);
    }

    // --- CROPPER MODAL LOGIC ---
    function openCropper(src) {
      cropTargetImage.src = src;
      cropperModal.style.display = 'flex';
      
      if (cropper) {
        cropper.destroy();
      }
      
      cropper = new Cropper(cropTargetImage, {
        aspectRatio: NaN, // Free ratio crop (Set e.g. 1 / 1 for square crop)
        viewMode: 1,
        autoCropArea: 0.9,
      });
    }

    function closeCropper() {
      cropperModal.style.display = 'none';
      if (cropper) {
        cropper.destroy();
        cropper = null;
      }
    }

    // Handle File Upload & Trigger Cropper Modal
    photoInput.addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = (event) => {
          rawImageSrc = event.target.result;
          openCropper(rawImageSrc);
        };
        reader.readAsDataURL(file);
      }
    });

    // Re-open cropper modal for current photo
    recropBtn.addEventListener('click', () => {
      if (rawImageSrc) {
        openCropper(rawImageSrc);
      }
    });

    // Apply Crop Selection
    applyCropBtn.addEventListener('click', () => {
      if (!cropper) return;
      const croppedCanvas = cropper.getCroppedCanvas();
      
      if (croppedCanvas) {
        const img = new Image();
        img.onload = () => {
          userImage = img;
          const aspect = img.width / img.height;
          baseWidth = 440;
          baseHeight = 440 / aspect;
          recropBtn.style.display = 'block';
          renderCanvas();
          closeCropper();
        };
        img.src = croppedCanvas.toDataURL('image/png');
      }
    });

    // Cancel Crop
    cancelCropBtn.addEventListener('click', closeCropper);

    // --- Drag and Drop Mouse / Touch Listeners ---
    function getCanvasCoordinates(e) {
      const rect = canvas.getBoundingClientRect();
      const clientX = e.touches ? e.touches[0].clientX : e.clientX;
      const clientY = e.touches ? e.touches[0].clientY : e.clientY;
      const scaleX = canvas.width / rect.width;
      const scaleY = canvas.height / rect.height;

      return {
        x: (clientX - rect.left) * scaleX,
        y: (clientY - rect.top) * scaleY
      };
    }

    function startDrag(e) {
      if (!userImage) return;
      isDragging = true;
      const coords = getCanvasCoordinates(e);
      dragStartX = coords.x - photoX;
      dragStartY = coords.y - photoY;
    }

    function doDrag(e) {
      if (!isDragging || !userImage) return;
      e.preventDefault();
      const coords = getCanvasCoordinates(e);
      
      let newX = coords.x - dragStartX;
      let newY = coords.y - dragStartY;

      if (newX > BOUNDS.MAX_X) newX = BOUNDS.MAX_X;
      if (newX < BOUNDS.MIN_X) newX = BOUNDS.MIN_X;
      if (newY > BOUNDS.MAX_Y) newY = BOUNDS.MAX_Y;
      if (newY < BOUNDS.MIN_Y) newY = BOUNDS.MIN_Y;

      photoX = newX;
      photoY = newY;

      renderCanvas();
    }

    function endDrag() {
      isDragging = false;
    }

    canvas.addEventListener('mousedown', startDrag);
    window.addEventListener('mousemove', doDrag);
    window.addEventListener('mouseup', endDrag);
    canvas.addEventListener('touchstart', startDrag);
    window.addEventListener('touchmove', doDrag, { passive: false });
    window.addEventListener('touchend', endDrag);

    // Zoom slider handler
    zoomInput.addEventListener('input', (e) => {
      scale = parseFloat(e.target.value);
      renderCanvas();
    });

    // Reset position button handler
    resetBtn.addEventListener('click', () => {
      photoX = 580;
      photoY = 410;
      scale = 1.58;
      zoomInput.value = 1.58;
      renderCanvas();
    });

    // Re-render live as text inputs change
    nameInput.addEventListener('input', renderCanvas);
    batchInput.addEventListener('input', renderCanvas);
    collegeInput.addEventListener('input', renderCanvas);

    // Initial render call
    renderCanvas();

    // Instant PNG Download at full 1080x1350 resolution
    downloadBtn.addEventListener('click', () => {
      const dataUrl = canvas.toDataURL('image/png');
      const a = document.createElement('a');
      a.href = dataUrl;
      a.download = `${nameInput.value || 'Fest-Registration'}.png`;
      a.click();
    });
  </script>
